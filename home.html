4 생산관리가 "노란 안내(입력 안 됨)" 뜰 때

생산실적 저장 표(production)를 아직 안 만든 것. 집·사무실 아무데서나 한 번만 하면 끝. 아래 SQL 복사 → Supabase → SQL Editor → 붙여넣기 → Run.

create table if not exists public.production (
  id bigint generated by default as identity primary key,
  pdate date not null default current_date,
  line text, item text,
  plan_qty numeric default 0, prod_qty numeric default 0, defect_qty numeric default 0,
  note text, src text, created_at timestamptz default now()
);
alter table public.production enable row level security;
drop policy if exists "anon all production" on public.production;
create policy "anon all production" on public.production for all to anon using (true) with check (true);
do $$ begin alter publication supabase_realtime add table production;
exception when duplicate_object then null; end $$;
